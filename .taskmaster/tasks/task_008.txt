   ```typescript
   // scoringService.ts
   import { db } from '../firebase/config';
   import { doc, updateDoc, getDoc } from 'firebase/firestore';
   
   export const updateScore = async (roundId: string, holeNumber: number, playerId: string, score: number) => {
     const roundRef = doc(db, 'rounds', roundId);
     const roundSnap = await getDoc(roundRef);
     if (!roundSnap.exists()) throw new Error('Round not found');
     
     const roundData = roundSnap.data();
     const scores = roundData.scores || {};
     
     // Update the score for this player and hole
     if (!scores[playerId]) scores[playerId] = {};
     scores[playerId][holeNumber] = score;
     
     await updateDoc(roundRef, { scores });
     return calculateGameResults(roundData.gameType, scores, roundData);
   };
   ```
